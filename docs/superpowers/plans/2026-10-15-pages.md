# Pages Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Facebook-style Pages: public presences with an avatar that users follow, an ADMIN/EDITOR team that posts as the page, and page posts that appear on the page and in followers' feeds.

**Architecture:** New `Page`, `PageMember`, `PageFollow` tables and a nullable `pageId` on `Post` (mutually exclusive with `groupId` via a DB CHECK). `canViewPost` gains a "page posts are public" branch; the feed query gains a "followed pages" OR branch; `PostCard` renders a page-authored post with the page's avatar and name. New `src/features/pages/` mirrors `src/features/groups/`; the slug helpers move to `src/lib/slug.ts` and are shared.

**Tech Stack:** Next.js 16 App Router (server components + server actions), TypeScript, Prisma 7 + PostgreSQL, zod 4, Tailwind, Vitest (pure units), Playwright (e2e, serial, against `pnpm dev`).

**Spec:** `docs/superpowers/specs/2026-10-15-pages-design.md`

## Global Constraints

- Follow `AGENTS.md`: this Next.js version differs from training data; consult `node_modules/next/dist/docs/` when unsure. Pages type props with the global `PageProps<'/route'>` helper; run `pnpm exec next typegen` if a new route's type is not yet known.
- Page name: 1–80 chars after trim. Description: ≤500 chars, empty stored as `null`. Slug: ≤60 chars, `[a-z0-9-]`, unique among pages, never changes after creation.
- Roles: exactly `ADMIN`, `EDITOR`. A page is never left without an ADMIN.
- Every page post is public to logged-in users. Page posts never appear on the acting user's profile. A post has at most one of `groupId` / `pageId`.
- Server actions return `ActionResult`; permission failures return `fail('...')`, never throw. All page-affecting actions revalidate with `revalidatePath('/', 'layout')`.
- After any `prisma migrate dev` / `prisma generate`, check `ls src/generated/prisma/models` contains `Page.ts`; if not, run `pnpm prisma generate`. Then restart the running `next dev` server (kill the node process on port 3000, `pnpm dev` again).

- Playwright runs with one worker; keep timeouts as configured.
- Commit messages end with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`. Do not commit `prisma/schema.sql`.

```ts
export type PostVisibilityInput = {
  authorId: string
  group: { id: string; privacy: GroupPrivacy } | null
  /** Set when the post was made on a page; page posts are public. */
  pageId: string | null
}

/**
 * Whether `viewerId` may see a post. Page posts are public. A group post is
 * visible when the group is public or the viewer is a member. A timeline post
 * follows the friends rule.
 */
export function canViewPost(
  viewerId: string,
  post: PostVisibilityInput,
  friendIds: readonly string[],
  memberGroupIds: readonly string[],
): boolean {
  if (post.pageId) return true
  if (post.group) {
    return (
      post.group.privacy === 'PUBLIC' || memberGroupIds.includes(post.group.id)
    )
  }
  return canViewPostsOf(viewerId, post.authorId, friendIds)
}
```

```ts
/** Page team role matrix; pure so it can be unit-tested and shared by server and client code. */

export const PAGE_ROLES = ['ADMIN', 'EDITOR'] as const
export type PageRole = (typeof PAGE_ROLES)[number]

export function isPageRole(value: unknown): value is PageRole {
  return (PAGE_ROLES as readonly unknown[]).includes(value)
}

/** Any team member posts as the page and may delete page posts. */
export function canPostAsPage(role: PageRole | null): boolean {
  return role === 'ADMIN' || role === 'EDITOR'
}

/** Admins only: settings, team, deleting the page. */
export function canManagePage(role: PageRole | null): boolean {
  return role === 'ADMIN'
}

type Actor = { id: string; role: PageRole }

/** Admins may remove anyone (themselves included); editors may only leave. */
export function canRemoveTeamMember(actor: Actor, targetId: string): boolean {
  return actor.role === 'ADMIN' || actor.id === targetId
}

export function canChangeTeamRole(actor: Actor): boolean {
  return actor.role === 'ADMIN'
}

/**
 * After removing (`newRole` null) or re-roling the target, is there still an
 * admin? Only touching the last ADMIN can break that.
 */
export function keepsAnAdmin(
  targetRole: PageRole,
  adminCount: number,
  newRole: PageRole | null,
): boolean {
  if (targetRole !== 'ADMIN') return true
  if (newRole === 'ADMIN') return true
  return adminCount > 1
}
```

```ts
import 'server-only'
import { prisma } from '@/lib/db'
import { personSelect, type Person } from '@/features/friends/queries'
import type { PageRole } from './permissions'

export const pageSelect = {
  id: true,
  slug: true,
  name: true,
  description: true,
  avatarUrl: true,
  _count: { select: { followers: true } },
} as const

export type PageSummary = {
  id: string
  slug: string
  name: string
  description: string | null
  avatarUrl: string | null
  followerCount: number
}

export type PageView = PageSummary & {
  viewerRole: PageRole | null
  following: boolean
}

export type TeamMember = { user: Person; role: PageRole; joinedAt: Date }

type PageRow = {
  id: string
  slug: string
  name: string
  description: string | null
  avatarUrl: string | null
  _count: { followers: number }
}

function toSummary(row: PageRow): PageSummary {
  return {
    id: row.id,
    slug: row.slug,
    name: row.name,
    description: row.description,
    avatarUrl: row.avatarUrl,
    followerCount: row._count.followers,
  }
}

/** Every page the user is on the team of, keyed by page id, with their role. */
export async function getPageRoles(
  userId: string,
): Promise<Map<string, PageRole>> {
  const rows = await prisma.pageMember.findMany({
    where: { userId },
    select: { pageId: true, role: true },
  })
  return new Map(rows.map((r) => [r.pageId, r.role]))
}

export async function getFollowedPageIds(userId: string): Promise<string[]> {
  const rows = await prisma.pageFollow.findMany({
    where: { userId },
    select: { pageId: true },
  })
  return rows.map((r) => r.pageId)
}

export async function getViewerPageRole(
  userId: string,
  pageId: string,
): Promise<PageRole | null> {
  const row = await prisma.pageMember.findUnique({
    where: { pageId_userId: { pageId, userId } },
    select: { role: true },
  })
  return row?.role ?? null
}

async function viewerState(
  viewerId: string,
  pageIds: string[],
): Promise<{ roles: Map<string, PageRole>; following: Set<string> }> {
  if (pageIds.length === 0) return { roles: new Map(), following: new Set() }
  const [members, follows] = await Promise.all([
    prisma.pageMember.findMany({
      where: { userId: viewerId, pageId: { in: pageIds } },
      select: { pageId: true, role: true },
    }),
    prisma.pageFollow.findMany({
      where: { userId: viewerId, pageId: { in: pageIds } },
      select: { pageId: true },
    }),
  ])
  return {
    roles: new Map(members.map((m) => [m.pageId, m.role])),
    following: new Set(follows.map((f) => f.pageId)),
  }
}

export async function getPageBySlug(
  viewerId: string,
  slug: string,
): Promise<PageView | null> {
  const row = await prisma.page.findUnique({
    where: { slug },
    select: pageSelect,
  })
  if (!row) return null
  const { roles, following } = await viewerState(viewerId, [row.id])
  return {
    ...toSummary(row),
    viewerRole: roles.get(row.id) ?? null,
    following: following.has(row.id),
  }
}

export async function getMyPages(
  userId: string,
): Promise<(PageSummary & { role: PageRole })[]> {
  const rows = await prisma.pageMember.findMany({
    where: { userId },
    select: { role: true, page: { select: pageSelect } },
    orderBy: { page: { name: 'asc' } },
  })
  return rows.map((r) => ({ ...toSummary(r.page), role: r.role }))
}

export async function getFollowedPages(userId: string): Promise<PageSummary[]> {
  const rows = await prisma.pageFollow.findMany({
    where: { userId },
    select: { page: { select: pageSelect } },
    orderBy: { page: { name: 'asc' } },
  })
  return rows.map((r) => toSummary(r.page))
}

/** Pages whose name contains `query` (case-insensitive), with the viewer's state for each. */
export async function searchPages(
  query: string,
  viewerId: string,
): Promise<PageView[]> {
  const q = query.trim()
  if (!q) return []
  const rows = await prisma.page.findMany({
    where: { name: { contains: q, mode: 'insensitive' } },
    select: pageSelect,
    orderBy: { name: 'asc' },
    take: 20,
  })
  const { roles, following } = await viewerState(
    viewerId,
    rows.map((r) => r.id),
  )
  return rows.map((row) => ({
    ...toSummary(row),
    viewerRole: roles.get(row.id) ?? null,
    following: following.has(row.id),
  }))
}

export async function getTeam(pageId: string): Promise<TeamMember[]> {
  const rows = await prisma.pageMember.findMany({
    where: { pageId },
```

```ts
function parsePage(formData: FormData) {

/** The avatar field, prepared for storage, or null when none was sent. */
async function prepareAvatar(formData: FormData) {
  const avatar = formData.get('avatar')
  if (!(avatar instanceof File) || avatar.size === 0) return { ok: true as const, upload: null }
  const prepared = await prepareUpload(avatar, 'avatars')
  if (!prepared.ok) return prepared
  return { ok: true as const, upload: prepared.upload }
}

export async function createPage(
  _prev: ActionResult,
  formData: FormData,
): Promise<ActionResult> {
  const user = await requireUser()
  const parsed = parsePage(formData)
  if (!parsed.success) return invalid(parsed)
  const avatar = await prepareAvatar(formData)
  if (!avatar.ok)
    return fail('Please fix the highlighted fields', { avatar: [avatar.error] })

  // Plain slug first; on a collision retry a few times with a random suffix.
  // Row(s) first, file second, inside one transaction (see createPost).
  const base = slugify(parsed.data.name)
  let slug: string | null = null
  for (let attempt = 0; attempt < 4 && !slug; attempt++) {
    const candidate = attempt === 0 ? base : withSuffix(base)
    try {
      await prisma.$transaction(async (tx) => {
        await tx.page.create({
          data: {
            ...parsed.data,
            slug: candidate,
            avatarUrl: avatar.upload?.url ?? null,
            members: { create: { userId: user.id, role: 'ADMIN' } },
            followers: { create: { userId: user.id } },
          },
        })
        if (avatar.upload) await writeUpload(avatar.upload)
      })
      slug = candidate
    } catch (e) {
      if (!isUniqueViolation(e)) throw e
    }
  }
  if (!slug)
    return fail('Could not pick a unique address for this page; try another name')

  revalidatePages()
  redirect(`/pages/${slug}`)
}

export async function updatePage(
  pageId: string,
  _prev: ActionResult,
  formData: FormData,
): Promise<ActionResult> {
  const user = await requireUser()
  if (!canManagePage(await getViewerPageRole(user.id, pageId)))
    return fail('Only admins can change page settings')
  const parsed = parsePage(formData)
  if (!parsed.success) return invalid(parsed)
  const avatar = await prepareAvatar(formData)
  if (!avatar.ok)
    return fail('Please fix the highlighted fields', { avatar: [avatar.error] })

  const previous = await prisma.page.findUnique({
    where: { id: pageId },
    select: { slug: true, avatarUrl: true },
  })
  if (!previous) return fail('Page not found')

  await prisma.$transaction(async (tx) => {
    await tx.page.update({

  })

export async function deletePage(pageId: string): Promise<ActionResult> {
  const user = await requireUser()
  if (!canManagePage(await getViewerPageRole(user.id, pageId)))
    return fail('Only admins can delete a page')

  const [page, images] = await Promise.all([
    prisma.page.findUnique({ where: { id: pageId }, select: { avatarUrl: true } }),
    prisma.post.findMany({
      where: { pageId, imageUrl: { not: null } },
      select: { imageUrl: true },
    }),
  ])
  if (!page) return fail('Page not found')
  // Row first, files second: a failed delete leaves no dangling references.
  await prisma.page.delete({ where: { id: pageId } })
  for (const { imageUrl } of images) await deleteUpload(imageUrl)
  await deleteUpload(page.avatarUrl)
  revalidatePages()
  return OK
}

async function withPageLock<T>(
  pageId: string,
  fn: (tx: Prisma.TransactionClient) => Promise<T>,

  return prisma.$transaction(async (tx) => {
    await tx.$executeRaw`SELECT 1 FROM "Page" WHERE id = ${pageId} FOR UPDATE`
    return fn(tx)
  })
}

): Promise<PageRole | null> {
  const row = await tx.pageMember.findUnique({
    where: { pageId_userId: { pageId, userId } },
    select: { role: true },
  })
  return row?.role ?? null
}

function adminCount(tx: Prisma.TransactionClient, pageId: string) {
  return tx.pageMember.count({ where: { pageId, role: 'ADMIN' } })
}

export async function addTeamMember(
  pageId: string,
  _prev: ActionResult,
  formData: FormData,
): Promise<ActionResult> {
  const user = await requireUser()
  if (!canManagePage(await getViewerPageRole(user.id, pageId)))
    return fail('Only admins can add team members')
  const username = String(formData.get('username') ?? '')
    .trim()
    .toLowerCase()
  const role = formData.get('role')
  if (!username) return fail('Enter a username')
  if (!isPageRole(role)) return fail('Choose a role')

  const target = await prisma.user.findUnique({
    where: { username },
    select: { id: true },
  })
  if (!target) return fail('No user with that username')
  const { count } = await prisma.pageMember.createMany({
    data: [{ pageId, userId: target.id, role }],
    skipDuplicates: true,
  })
  if (count === 0) return fail('Already on the team')
  revalidatePages()
  return OK
}

export async function setTeamRole(
  pageId: string,
  userId: string,
  role: string,
): Promise<ActionResult> {
  const user = await requireUser()
  if (!isPageRole(role)) return fail('Unknown role')
  const result = await withPageLock(pageId, async (tx) => {
    const actorRole = await roleIn(tx, pageId, user.id)
    if (!actorRole || !canChangeTeamRole({ id: user.id, role: actorRole }))
      return fail('Only admins can change roles')
    const targetRole = await roleIn(tx, pageId, userId)
    if (!targetRole) return fail('That person is not on the team')
    if (!keepsAnAdmin(targetRole, await adminCount(tx, pageId), role))
      return fail('Add another admin first')
    await tx.pageMember.update({
      where: { pageId_userId: { pageId, userId } },
      data: { role },
    })
    return OK
  })
  if (result.ok) revalidatePages()
  return result
}

export async function removeTeamMember(
  pageId: string,
  userId: string,
): Promise<ActionResult> {
  const user = await requireUser()
  const result = await withPageLock(pageId, async (tx) => {
    const actorRole = await roleIn(tx, pageId, user.id)
    if (!actorRole) return fail('You are not on this team')
    if (!canRemoveTeamMember({ id: user.id, role: actorRole }, userId))
      return fail('Only admins can remove other team members')
    const targetRole = await roleIn(tx, pageId, userId)
    if (!targetRole) return fail('That person is not on the team')
    if (!keepsAnAdmin(targetRole, await adminCount(tx, pageId), null))
      return fail('Add another admin first')
    await tx.pageMember.deleteMany({ where: { pageId, userId } })
    return OK
  })
  if (result.ok) revalidatePages()
  return result
}
```

```ts
async function createPage(
  page: Page,
  name: string,
  withAvatar = false,
): Promise<string> {
  await page.goto('/pages/new')
  await page.getByLabel('Page name').fill(name)
  await page.getByLabel('Description').fill(`About ${name}`)
  if (withAvatar) {
    await page
      .getByLabel('Avatar')
      .setInputFiles({ name: 'a.png', mimeType: 'image/png', buffer: PNG_1x1 })
  }
  await page.getByRole('button', { name: 'Create page' }).click()
  await expect(page).toHaveURL(/\/pages\/[a-z0-9-]+$/)
  await expect(page.getByRole('heading', { level: 1 })).toContainText(name)
  return page.url()
}

async function postAs(page: Page, body: string) {
  await page.getByLabel("What's on your mind?").fill(body)
  await page.getByRole('button', { name: 'Post' }).click()
  await expect(page.getByRole('article').filter({ hasText: body })).toHaveCount(1)
}

async function addToTeam(admin: Page, url: string, username: string, role: 'Editor' | 'Admin') {
  await admin.goto(`${url}/team`)
  await admin.getByLabel('Username').fill(username)
  await admin.getByLabel('Role', { exact: true }).selectOption({ label: role })
  await admin.getByRole('button', { name: 'Add to team' }).click()
  await expect(
    admin.getByRole('region', { name: 'Team' }).getByRole('listitem'),

test('create a page with an avatar; it is listed and the avatar is served', async ({
  browser,
```
